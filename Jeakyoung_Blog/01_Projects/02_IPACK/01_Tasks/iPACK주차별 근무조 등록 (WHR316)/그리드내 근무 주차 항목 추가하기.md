다음과 같이 년-월-일 형식으로 그리드내에 출력해줍니다.

```javascript
WHR316_VW_btnCng: function (btn) {
        var me = this;
        var grid = me.getView().lookup('grid');
        var store = grid.getStore();

        store.each(function (rec) {
            var wgCode = rec.get('WG_CODE');
            if (rec.get('WG_CODE')) {
                rec.set('FACTORY_CODE', me.getView().lookup('combo_01').getValue());
                rec.set('OPMAN_CODE', gUserId);
                rec.set('WG_FROM_DATE', me.getView().lookup('combo_05').getValue());
            }
            if (wgCode === '001') {
                rec.set('WG_CODE', '002');
            } else if (wgCode === '002') {
                rec.set('WG_CODE', '001');
            }
        });

        store.sync({
            success: function () {
                showToast('알림', '근무조를 변환하였습니다.');
                me.onView();
            },
        });
    },
```

해당 기능 구현시 발견하게된 오류 사항으로 변환 클릭시 적용할필요가 없는 행에도 근무주차 시작, 마감일을 등록하는것

을 확인했습니다.

이를 해결하기위해 if절을 추가하여 WG_CODE가 있는경우에만 업데이트를 진행하게 됩니다.

## ✅ 검증 및 결과

- [x] `WHR316_ST.js`(6~11행)의 `store_01.fields`에 `WG_FROM_DATE`/`WG_TO_DATE`가 `dateReadFormat: 'Ymd'`로 실제 존재.
- [x] `WHR316_CT.js`의 `WHR316_VW_btnCng`(393~430행)에 `if (wgCode && rec.get('SELECT_FLAG') === true)` 조건이 남아있어, "WG_CODE 있는 행에만 반영" 수정이 최종 코드에도 반영돼 있음.
