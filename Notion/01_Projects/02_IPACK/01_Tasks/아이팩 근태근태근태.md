    convertToMinusHour: function (rawTime) {
        if (rawTime === null || rawTime === undefined || rawTime === '') return NaN;
        if (String(rawTime).indexOf(':') >= 0) {
            var parts = String(rawTime).split(':');
            if (parts.length >= 2) {
                var h = parseInt(parts[0], 10) || 0;
                var m = parseInt(parts[1], 10) || 0;
                var total = h + (Math.ceil(m / 10) * 10) / 60;
                return Number(total.toFixed(2));
            }
        }
        var s = String(rawTime).replace(':', '');
        if (s.length === 3) s = '0' + s;
        if (!/^\d+$/.test(s)) return NaN;
        var num = parseFloat(s) / 100.0;
        var hours = Math.floor(num);
        var minutes = Math.ceil((num - hours) * 100); //아랫값 전부 ceil로 (지각 조퇴)
        minutes = Math.ceil(minutes / 10) * 10;
        if (minutes === 60) {
            hours += 1;
            minutes = 0;
        }
        return Number((hours + minutes / 60).toFixed(2));
    },
