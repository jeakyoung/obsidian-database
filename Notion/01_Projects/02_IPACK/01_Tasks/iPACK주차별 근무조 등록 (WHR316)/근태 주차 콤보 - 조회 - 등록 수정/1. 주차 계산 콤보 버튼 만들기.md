## DATEPART

```sql
1. 해당 날짜가 소속된 주가 해당 연도의 몇주차인지?

DECLARE @YYYYMMDD AS VARCHAR(8)='20180308' 

SELECT LEFT(@YYYYMMDD, 4) AS 년도, DATEPART(WK, CONVERT(DATE,@YYYYMMDD)) AS 주차 





2. 해당 주차의 시작일과 종료일을 각각 8자리 문자로

DECLARE @YYYYMMDD AS VARCHAR(8)='20180308' 

SELECT  CONVERT(VARCHAR(8),DATEADD(DAY, 1 - DATEPART(DW, CONVERT(DATE,@YYYYMMDD)) , CONVERT(DATE,@YYYYMMDD)),112) AS 시작일자 

,     CONVERT(VARCHAR(8),DATEADD(DAY, 7 - DATEPART(DW, CONVERT(DATE,@YYYYMMDD)) , CONVERT(DATE,@YYYYMMDD)),112) AS 종료일자 





3. 전년도 동 주차의 시작일과 종료일을 각각 8자리 문자로 

DECLARE @YYYYMMDD AS VARCHAR(8)='20180308' 

DECLARE @PREV_YYYY AS VARCHAR(4) --전년도 

DECLARE @WK_NUM AS INT --주차 

SELECT @PREV_YYYY = CONVERT(VARCHAR(4),CONVERT(INT,LEFT(@YYYYMMDD, 4)-1))  

,    @WK_NUM   = DATEPART(WK, CONVERT(DATE,@YYYYMMDD))  

;  

WITH DATE_LIST  

AS  

(  

   SELECT CAST (@PREV_YYYY+'-01-01' AS DATETIME) AS DT  

   UNION ALL  

   SELECT DT+1  

   FROM   DATE_LIST  

   WHERE DT+1<=@PREV_YYYY+'-12-31'  

)  

-- MAXRECURSION : 서버차원의기본값은100입니다. 0을지정하면제한이적용되지않습니다  

SELECT CONVERT(VARCHAR(8),MIN(DT),112) AS 전년도시작일자 

,      CONVERT(VARCHAR(8),MAX(DT),112) AS 전년도종료일자 

FROM DATE_LIST   

WHERE DATEPART(wk, DT) = @WK_NUM  

OPTION(MAXRECURSION 0)  
```

해당 기능을 구현하기 위해 사용한 DATEPART 레퍼런스입니다.

## 코드 내용

```sql
USE [iPlusERP_IPACK_DEV]
GO
/****** Object:  StoredProcedure [dbo].[SP_WHR316_COMBO_01]    Script Date: 2025-09-12 오전 9:40:59 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
ALTER PROC [dbo].[SP_WHR316_COMBO_01]
(
    @FACTORY_CODE VARCHAR(6),
    @YYYYMM       CHAR(6)
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @STARTDATE DATE = CONVERT(DATE, @YYYYMM + '01');
    --월정보에 1일을 적재 (9월로 들어오면 9월1일을 STARTDATE로)
    DECLARE @ENDDATE DATE = EOMONTH(@STARTDATE);
    --시작일을 기준으로 끝날짜를 찾기

    DECLARE @WEEKS TABLE (
    --임시 테이블 선언
        WG_FROM_DATE CHAR(8),
        DISPLAY_TEXT VARCHAR(20)
        --데이터 출력형식
    );

    DECLARE @WEEKSTART DATE;
    DECLARE @WEEKEND DATE;

    SET @WEEKSTART = DATEADD(DAY, 1 - DATEPART(WEEKDAY, @STARTDATE), @STARTDATE);
    --월의 시작일을 바탕으로 주차시작일만 미리계산

    WHILE @WEEKSTART <= @ENDDATE
    --월의 끝날짜 까지
    BEGIN
        SET @WEEKEND = DATEADD(DAY, 6, @WEEKSTART);
        --시작일을 가지고 6일후로 설정 (주차종료일)

        BEGIN
            INSERT INTO @WEEKS
            --이걸 전부 임시테이블로 저장
			(
			  WG_FROM_DATE
			, DISPLAY_TEXT
			)
            VALUES (
                CONVERT(CHAR(8), @WEEKSTART, 112),
                FORMAT(@WEEKSTART, 'MM-dd') + ' ~ ' + FORMAT(@WEEKEND, 'MM-dd')
                --출력포맷을 원하는 형태로 변환
            )
        END

        SET @WEEKSTART = DATEADD(DAY, 7, @WEEKSTART);
    END

    SELECT 
	  WG_FROM_DATE
	, DISPLAY_TEXT
    FROM @WEEKS
		--임시 테이블내 데이터를 전부 SELECT
END

```
