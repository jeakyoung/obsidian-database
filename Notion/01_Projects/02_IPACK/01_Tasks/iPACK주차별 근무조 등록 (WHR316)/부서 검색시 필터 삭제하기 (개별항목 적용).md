```javascript
onComboChange: function (dv, newValue, oldValue, eOpts) {
        var me = this;

        switch (dv.itemId) {
            case 'cbo_Factory': {
                me.onView();
                break;
            }
            case 'columns_tf01': {
                if (newValue) {
                    dv.getTrigger('clear').show();
                } else {
                    dv.getTrigger('clear').hide();
                }
            }
```
