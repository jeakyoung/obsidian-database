USE [iPlusERP_IPACK_DEV]
GO

ALTER PROCEDURE [dbo].[SP_WHR328_02_LIST]          
(               
	@FACTORY_CODE		VARCHAR(6) 
	,@SALARY_MONTH		VARCHAR(6)
	,@EMPLOYEE_NO		VARCHAR(8)	
)
AS  
BEGIN              
	SET NOCOUNT ON;  
--
	
	CREATE TABLE #TEMP_CALENDAR
	(FACTORY_CODE  VARCHAR(6)
	,HD_DATE VARCHAR(8)
	,HD_GBN VARCHAR(3))       

	DECLARE @S_DATE VARCHAR(10)
	DECLARE @E_DATE VARCHAR(10)

	SET @S_DATE = LEFT(@SALARY_MONTH,4) + SUBSTRING(@SALARY_MONTH,5,2) + '01'
	SET @E_DATE = CONVERT(CHAR(8), DATEADD(DAY, -1, DATEADD(MONTH, 1, LEFT(@SALARY_MONTH,4) + '-' + SUBSTRING(@SALARY_MONTH,5,2) + '-01')),112)
	
	--SELECT	@S_DATE
	--	,	@E_DATE 

	WHILE @S_DATE <= @E_DATE

	BEGIN
		INSERT #TEMP_CALENDAR 
		VALUES(
			@FACTORY_CODE
			,@S_DATE
			,DATEPART(DW, LEFT(@S_DATE,4) + '-' + SUBSTRING(@S_DATE,5,2) + '-' + RIGHT(@S_DATE,2)) /*DATE에 해당하는 요일넘버, 1(일요일)~7(토요일)*/
			)
		SET @S_DATE = CONVERT(CHAR(8),DATEADD(D,1,@S_DATE),112)
	END	

	;WITH Event_CTE AS (
       SELECT 
           EMPLOYEE_NO,
           KT_DATETIME,
           NREADERIDN,
           CASE 
             WHEN NREADERIDN IN (546839596, 546839166) THEN '입실'
             WHEN NREADERIDN IN (546839433, 546839434) THEN '퇴실'
             ELSE '기타'
           END AS IO_NAME,
           CONVERT(CHAR(8), KT_DATETIME, 112) AS WORK_DATE
       FROM TIN334
       WHERE EMPLOYEE_NO = @EMPLOYEE_NO
         AND CONVERT(CHAR(6), KT_DATETIME, 112) = @SALARY_MONTH
         AND NREADERIDN IN (546839166, 546839433, 546839434, 546839596)
   ),

   InOut AS (
       SELECT 
           EMPLOYEE_NO,
           WORK_DATE,
           MIN(CASE WHEN IO_NAME='입실' THEN KT_DATETIME END) AS FIRST_IN,
           MAX(CASE WHEN IO_NAME='퇴실' THEN KT_DATETIME END) AS LAST_OUT
       FROM Event_CTE
       GROUP BY EMPLOYEE_NO, WORK_DATE
	   )
   --,

   --OutSum AS (
   --    SELECT 
   --        O.EMPLOYEE_NO, 
   --        CONVERT(CHAR(8), O.KT_DATETIME, 112) AS WORK_DATE,
   --        SUM(DATEDIFF(MINUTE, O.KT_DATETIME, I.KT_DATETIME)) AS TOTAL_OUT_MINUTES
   --    FROM Event_CTE O
   --    JOIN Event_CTE I 
   --      ON O.EMPLOYEE_NO = I.EMPLOYEE_NO 
   --     AND I.KT_DATETIME > O.KT_DATETIME
   --    WHERE O.IO_NAME='퇴실' AND I.IO_NAME='입실'
   --    GROUP BY O.EMPLOYEE_NO, CONVERT(CHAR(8), O.KT_DATETIME, 112)
   --)
	
	  
	 SELECT 	ISNULL(SUBSTRING(A.HD_DATE,7,2) + CASE WHEN DATEPART (WEEKDAY , A.HD_DATE) = '1' THEN '(일)'
											 WHEN DATEPART (WEEKDAY , A.HD_DATE) = '2' THEN '(월)'
											 WHEN DATEPART (WEEKDAY , A.HD_DATE) = '3' THEN '(화)'
											 WHEN DATEPART (WEEKDAY , A.HD_DATE) = '4' THEN '(수)'
											 WHEN DATEPART (WEEKDAY , A.HD_DATE) = '5' THEN '(목)'
											 WHEN DATEPART (WEEKDAY , A.HD_DATE) = '6' THEN '(금)'
											 ELSE '(토)' END, '합계') AS  DU_YMD_DATE

	 , ISNULL(A.HD_DATE,'')			AS HD_DATE
	 , ISNULL(B.DU_YMD,'')			AS DU_YMD
	 , ISNULL(B.WG_CODE,'')			AS WG_CODE
	 , ISNULL(B.WG_PRDFROM,'')		AS WG_PRDFROM
	 , ISNULL(B.WG_PRDFROM2,'')		AS WG_PRDFROM2
	 , ISNULL(B.WG_PRDTO,'')		AS WG_PRDTO
	 , ISNULL(B.WG_PRDTO2,'')		AS WG_PRDTO2
	 , ISNULL(B.CODE_NAME_FULL,'')	AS CODE_NAME_FULL
	 --, CASE WHEN ISNULL(B.RESET_SUME_TIME,'') = '' THEN 0 ELSE B.RESET_SUME_TIME END AS RESET_SUME_TIME
	 , CASE WHEN ISNULL(B.NIGHT_SUM_TIME,'') = ''  THEN 0 ELSE B.NIGHT_SUM_TIME END AS NIGHT_SUM_TIME
	 , CASE WHEN ISNULL(B.ADD_SUM_TIME,'') = ''	   THEN 0 ELSE B.ADD_SUM_TIME END AS ADD_SUM_TIME
	 , CASE WHEN ISNULL(B.JI_SUM_TIME,'') = ''	   THEN 0 ELSE B.JI_SUM_TIME END AS JI_SUM_TIME
	 , CASE WHEN ISNULL(B.JO_SUM_TIME,'') = ''	   THEN 0 ELSE B.JO_SUM_TIME END AS JO_SUM_TIME
	 , CASE WHEN ISNULL(B.OI_SUM_TIME,'') = ''	   THEN 0 ELSE B.OI_SUM_TIME END AS OI_SUM_TIME
	 --, CASE WHEN ISNULL(B.RESET_SUME_CNT,'') = ''  THEN 0 ELSE B.RESET_SUME_CNT END AS RESET_SUME_CNT
	 , CASE WHEN ISNULL(B.NIGHT_SUM_CNT,'')	= ''   THEN 0 ELSE B.NIGHT_SUM_CNT END AS NIGHT_SUM_CNT
	 , CASE WHEN ISNULL(B.ADD_SUM_CNT,'') = ''     THEN 0 ELSE B.ADD_SUM_CNT END AS ADD_SUM_CNT
	 , CASE WHEN ISNULL(B.JI_SUM_CNT,'') = ''      THEN 0 ELSE B.JI_SUM_CNT END AS JI_SUM_CNT
	 , CASE WHEN ISNULL(B.JO_SUM_CNT,'') = ''      THEN 0 ELSE B.JO_SUM_CNT END AS JO_SUM_CNT
	 , CASE WHEN ISNULL(B.OI_SUM_CNT,'') = ''      THEN 0 ELSE B.OI_SUM_CNT END AS OI_SUM_CNT
	 , CASE WHEN ISNULL(I.FIRST_IN, '') = ''	   THEN '' ELSE FORMAT(I.FIRST_IN, 'HH:mm') END AS FIRST_IN_TIME
     , CASE WHEN ISNULL(I.LAST_OUT, '') = ''	   THEN '' ELSE FORMAT(I.LAST_OUT, 'HH:mm') END AS LAST_OUT_TIME
     --, CAST(ROUND(ISNULL(O.TOTAL_OUT_MINUTES,0) / 60.0 / 60.0, 2) AS DECIMAL(10,2)) AS OUT_HOURS
     , CAST(ROUND(ISNULL(DATEDIFF(MINUTE, I.FIRST_IN, I.LAST_OUT),0) / 60.0, 2) AS DECIMAL(10,2)) AS WORK_HOURS
	 FROM #TEMP_CALENDAR A		
		LEFT JOIN (
			SELECT A.EMPLOYEE_NO
				,	A.DU_YMD
				,   W.WG_CODE
				,	A.WG_PRDFROM
				,	A.WG_PRDFROM2
				,	A.WG_PRDTO
				,	A.WG_PRDTO2
				,	A.WG_ATTEND_GBN
				,	D.CODE_NAME_FULL

				--,	SUM(CASE WHEN F.DS_CODE = '24' THEN F.DS_TIME ELSE 0 END ) AS RESET_SUME_TIME
				--,	SUM(CASE WHEN F.DS_CODE = '37' THEN F.DS_TIME ELSE 0 END ) AS NIGHT_SUM_TIME
				--,	SUM(CASE WHEN F.DS_CODE = '33' THEN F.DS_TIME ELSE 0 END ) AS ADD_SUM_TIME
				--,	SUM(CASE WHEN F.DS_CODE = '10' THEN F.DS_TIME ELSE 0 END ) AS JI_SUM_TIME
				--,	SUM(CASE WHEN F.DS_CODE = '12' THEN F.DS_TIME ELSE 0 END ) AS JO_SUM_TIME
				--,	SUM(CASE WHEN F.DS_CODE = '38' THEN F.DS_TIME ELSE 0 END ) AS OI_SUM_TIME

				--,	SUM(CASE WHEN F.DS_CODE = '24' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS RESET_SUME_CNT
				--,	SUM(CASE WHEN F.DS_CODE = '37' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS NIGHT_SUM_CNT
				--,	SUM(CASE WHEN F.DS_CODE = '33' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS ADD_SUM_CNT
				--,	SUM(CASE WHEN F.DS_CODE = '10' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS JI_SUM_CNT
				--,	SUM(CASE WHEN F.DS_CODE = '12' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS JO_SUM_CNT
				--,	SUM(CASE WHEN F.DS_CODE = '38' AND F.DS_TIME > 0 THEN 1 ELSE 0 END ) AS OI_SUM_CNT

				--,	SUM(CASE WHEN C.DS_CODE = '24' THEN C.WG_APPPRD ELSE 0 END ) AS RESET_SUME_TIME
				,	MAX(CASE WHEN C.DS_CODE = '37' THEN CONVERT(FLOAT, F.DS_TIME) ELSE 0 END ) AS NIGHT_SUM_TIME
				,	SUM(CASE WHEN C.DS_CODE = '33' THEN CONVERT(FLOAT, F.DS_TIME) ELSE 0 END ) AS ADD_SUM_TIME
				,	SUM(CASE WHEN C.DS_CODE = '10' THEN CONVERT(FLOAT, F.DS_TIME) ELSE 0 END ) AS JI_SUM_TIME
				,	SUM(CASE WHEN C.DS_CODE = '12' THEN CONVERT(FLOAT, F.DS_TIME) ELSE 0 END ) AS JO_SUM_TIME
				,	SUM(CASE WHEN C.DS_CODE = '38' THEN CONVERT(FLOAT, F.DS_TIME) ELSE 0 END ) AS OI_SUM_TIME

				--,	SUM(CASE WHEN C.DS_CODE = '24' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS RESET_SUME_CNT
				,	SUM(CASE WHEN C.DS_CODE = '37' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS NIGHT_SUM_CNT
				,	SUM(CASE WHEN C.DS_CODE = '33' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS ADD_SUM_CNT
				,	SUM(CASE WHEN C.DS_CODE = '10' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS JI_SUM_CNT
				,	SUM(CASE WHEN C.DS_CODE = '12' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS JO_SUM_CNT
				,	SUM(CASE WHEN C.DS_CODE = '38' AND C.WG_APPPRD > 0 THEN 1 ELSE 0 END ) AS OI_SUM_CNT
				
			FROM TIN313 A				
				LEFT JOIN TCO101 D
				ON A.WG_ATTEND_GBN = D.CODE_ID2
				AND D.CODE_ID1 = '256'
				LEFT JOIN TIN312W W
				ON A.EMPLOYEE_NO = W.EMPLOYEE_NO
				AND A.DU_YMD BETWEEN W.WG_FROM_DATE AND W.WG_TO_DATE
				LEFT JOIN TIN514 F
				ON A.EMPLOYEE_NO = F.Du_Sbn
				LEFT JOIN TIN313D C
				ON A.EMPLOYEE_NO = C.EMPLOYEE_NO
				AND A.DU_YMD = C.DU_YMD
				AND C.WG_APPPRD > 0
				
				
			WHERE A.EMPLOYEE_NO = @EMPLOYEE_NO

			GROUP BY A.EMPLOYEE_NO
				,	A.DU_YMD
				,	W.WG_CODE
				,	A.WG_PRDFROM
				,	A.WG_PRDFROM2
				,	A.WG_PRDTO
				,	A.WG_PRDTO2
				,	A.WG_ATTEND_GBN
				,	D.CODE_NAME_FULL
			)B
			ON A.HD_DATE = B.DU_YMD
			LEFT JOIN InOut I ON A.HD_DATE = I.WORK_DATE
			--LEFT JOIN OutSum O ON A.HD_DATE = O.WORK_DATE
			
	ORDER BY A.HD_DATE
END
