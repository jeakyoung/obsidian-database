USE [iPlusERP_SJ]
GO
/****** Object:  StoredProcedure [dbo].[SP_WMA624_01_IUD]    Script Date: 2026-06-08 오전 10:44:37 ******/

ALTER PROC [dbo].[SP_WMA624_01_IUD]
(
	  @FACTORY_CODE				VARCHAR(6)
	, @IO_NO					VARCHAR(12)
	, @IO_FLAG					VARCHAR(3)
	, @ACC_FLAG					VARCHAR(6)
	, @MATERIAL_CODE			VARCHAR(30)
	, @ALTER_MATERIAL_CODE		VARCHAR(30)
	, @IO_DATE					VARCHAR(8)
	, @IO_CHARGE				VARCHAR(10)
	, @IO_QTY					FLOAT
	, @OUT_WAREHOUSE_CODE		VARCHAR(6)
	, @OUT_LOCATION_CODE		VARCHAR(6)
	, @IN_WAREHOUSE_CODE		VARCHAR(6)
	, @IN_LOCATION_CODE			VARCHAR(6)
	, @REMARK					VARCHAR(255)
	, @OPMAN_CODE				VARCHAR(10)
	, @SYSTEM_LOT_NO			VARCHAR(30)
	, @ALTER_SYSTEM_LOT_NO		VARCHAR(30)
	, @EXPIRE_DATE				VARCHAR(8)
	, @MANUFACTURE_DATE			VARCHAR(8)
	, @PLANT_CODE				VARCHAR(10) = ''
	, @IRU						CHAR(2)
	
 )
  
 AS 

 BEGIN 
    DECLARE @OPTIME  VARCHAR(12),
            @CNT     INT,
            @ERR_MSG VARCHAR(1000)
	SET @OPTIME = REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR(16), GETDATE(), 120), '-', ''), ':',''), ' ', '');

	IF @IO_NO = ''
	
		BEGIN
			DECLARE @DATE VARCHAR(12)
			SET @DATE = REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR(10), GETDATE(), 120), '-', ''), ':',''), ' ', '');			
			SELECT	@IO_NO = @DATE + RIGHT ('0000'+LTRIM(STR(ISNULL(MAX(SUBSTRING(IO_NO,9,4)), 0) + 1)),4)
			FROM	TMA536 with(nolock) 
			WHERE	SUBSTRING(IO_NO,1,8) = @DATE

		END
	

	IF @IRU = 'IU' 
		BEGIN 

        IF @IO_FLAG = '2'
        BEGIN

        BEGIN TRY 
        SELECT @CNT = COUNT(*)
	    FROM ESMMTGL AS A
            
	        -- 재고량 정보 JOIN
            LEFT JOIN (
            SELECT SUM(MAT.IN_QTY) - SUM(MAT.OUT_QTY) AS STOCK_QTY,
                MAT.FACTORY_CODE,
                MAT.PLANT_CODE,
                MAT.WAREHOUSE_CODE,
                MAT.MATERIAL_CODE,
                MAT.SYSTEM_LOT_NO
                FROM TMA922M MAT
                GROUP BY MAT.FACTORY_CODE, MAT.PLANT_CODE, MAT.WAREHOUSE_CODE, MAT.MATERIAL_CODE, MAT.SYSTEM_LOT_NO
            ) AS MAT
            ON MAT.FACTORY_CODE = '000001'
            AND MAT.PLANT_CODE = @PLANT_CODE 
            AND MAT.WAREHOUSE_CODE = @OUT_WAREHOUSE_CODE 
            AND A.ITEM_CD = MAT.MATERIAL_CODE
            AND MAT.SYSTEM_LOT_NO = @SYSTEM_LOT_NO

            LEFT JOIN(
			SELECT	A.FACTORY_CODE
			,	A.PLANT_CODE
			,	A.WAREHOUSE_CODE
			,	A.SYSTEM_LOT_NO
			,	A.MATERIAL_CODE
			,	SUM(A.IN_QTY) - SUM(A.OUT_QTY) AS TARGET_QTY
		    FROM	TMA922 A
		    WHERE	A.PLANT_CODE = @PLANT_CODE 
		    AND		(@OUT_WAREHOUSE_CODE = '' or (@OUT_WAREHOUSE_CODE <> '' and A.WAREHOUSE_CODE = @OUT_WAREHOUSE_CODE))
		    AND		A.IO_DATE <= @IO_DATE
            AND     A.SYSTEM_LOT_NO = @SYSTEM_LOT_NO
		    GROUP BY	A.FACTORY_CODE
			        ,	A.PLANT_CODE
			        ,	A.WAREHOUSE_CODE
			        ,	A.SYSTEM_LOT_NO
			        ,	A.MATERIAL_CODE
		    ) AS B
            ON MAT.FACTORY_CODE = B.FACTORY_CODE
            AND A.ITEM_CD = B.MATERIAL_CODE
            AND A.WERKS = B.PLANT_CODE

            LEFT JOIN TCO102 C
		    ON  MAT.FACTORY_CODE = C.FACTORY_CODE
		    AND MAT.PLANT_CODE = C.PLANT_CODE
            AND MAT.WAREHOUSE_CODE = C.WAREHOUSE_CODE 

	    WHERE A.WERKS = @PLANT_CODE 
	    and A.ITEM_CD = @MATERIAL_CODE
	    and MAT.WAREHOUSE_CODE =@OUT_WAREHOUSE_CODE 
        AND MAT.SYSTEM_LOT_NO = @SYSTEM_LOT_NO
        --AND ((B.TARGET_QTY > @IO_QTY) AND MAT.STOCK_QTY - (B.TARGET_QTY - @IO_QTY) < 0)
        AND MAT.STOCK_QTY < @IO_QTY


        IF @CNT = 1
        BEGIN
            SET @ERR_MSG = '[LOT] ' + @SYSTEM_LOT_NO + ' 현재재고가 변경되어 수량이 부족합니다.' 
            RAISERROR (@ERR_MSG,16,1) with log; 
        END
        END TRY

            
        BEGIN CATCH
            SET @ERR_MSG = ERROR_MESSAGE() 
            RAISERROR (@ERR_MSG,16,1) with log; 
        END CATCH

        END
			UPDATE TMA536 SET 
				IO_FLAG					=		@IO_FLAG
				,	ACC_FLAG			=		@ACC_FLAG
				,	MATERIAL_CODE		=		@MATERIAL_CODE
				,	ALTER_MATERIAL_CODE	=		@ALTER_MATERIAL_CODE
				,	IO_DATE				=		@IO_DATE
				,	IO_CHARGE			=		@IO_CHARGE
				,	IO_QTY				=		@IO_QTY
				,	OUT_WAREHOUSE_CODE	=		@OUT_WAREHOUSE_CODE
				,	OUT_LOCATION_CD		=		@OUT_LOCATION_CODE
				,	IN_WAREHOUSE_CODE	=		@IN_WAREHOUSE_CODE
				,	IN_LOCATION_CD		=		@IN_LOCATION_CODE
				,	REMARK				=		@REMARK
				,	OPMAN_CODE2			=		@OPMAN_CODE
				,	OPTIME2				=		@OPTIME
				,	SYSTEM_LOT_NO		=		@SYSTEM_LOT_NO
				,	ALTER_SYSTEM_LOT_NO	=		@ALTER_SYSTEM_LOT_NO
			WHERE FACTORY_CODE		=		@FACTORY_CODE
			AND IO_NO				=		@IO_NO

			IF	@@ROWCOUNT	=	0 
				BEGIN

					--DECLARE @DATE VARCHAR(12)
					--SET @DATE = REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR(10), GETDATE(), 120), '-', ''), ':',''), ' ', '');

					--SELECT	@IO_NO = @DATE + RIGHT ('0000'+LTRIM(STR(ISNULL(MAX(SUBSTRING(IO_NO,9,4)), 0) + 1)),4)
					--FROM	TMA536
					--WHERE	SUBSTRING(IO_NO,1,8) = @DATE

					INSERT INTO TMA536
					( 
						FACTORY_CODE
						,IO_NO
						,IO_FLAG
						,ACC_FLAG
						,MATERIAL_CODE
						,ALTER_MATERIAL_CODE
						,IO_DATE
						,IO_CHARGE
						,IO_QTY
						,OUT_WAREHOUSE_CODE
						,OUT_LOCATION_CD
						,IN_WAREHOUSE_CODE
						,IN_LOCATION_CD
						,REMARK
						,OPMAN_CODE
						,OPTIME
						,SYSTEM_LOT_NO
						,ALTER_SYSTEM_LOT_NO
						,EXPIRE_DATE
						,MANUFACTURE_DATE
						--, PLANT_CODE
					) 
					VALUES 
					( 
						@FACTORY_CODE
						,@IO_NO
						,@IO_FLAG
						,@ACC_FLAG
						,@MATERIAL_CODE
						,@ALTER_MATERIAL_CODE
						,@IO_DATE
						,@IO_CHARGE
						,@IO_QTY
						,@OUT_WAREHOUSE_CODE
						,@OUT_LOCATION_CODE
						,@IN_WAREHOUSE_CODE
						,@IN_LOCATION_CODE
						,@REMARK
						,@OPMAN_CODE
						,@OPTIME
						,@SYSTEM_LOT_NO
						,@ALTER_SYSTEM_LOT_NO
						,@EXPIRE_DATE
						,@MANUFACTURE_DATE
						--, @PLANT_CODE
					) 
				END	
		END		
	ELSE	
		BEGIN		
			DELETE FROM TMA536
			WHERE	FACTORY_CODE	=	@FACTORY_CODE
			AND				IO_NO	=	@IO_NO
		END 
 END
