Next 버튼 동작 CT → Month를 가져와서 → +1

```sql
USE [iPlusERP_IPACK_DEV]
GO
/****** Object:  StoredProcedure [dbo].[SP_WHR316_01_LIST]    Script Date: 2025-09-10 오전 9:59:00 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

ALTER PROC	[dbo].[SP_WHR316_01_LIST]
(
		@FACTORY_CODE		VARCHAR(6)
	 ,	@PAY_GROUP_CODE		VARCHAR(10)
	 ,	@PAY_TYPE_CODE		VARCHAR(10)
	 ,	@WG_CODE			VARCHAR(6)
	 ,  @DEPT_NAME			VARCHAR(10)
	 ,	@BASE_NAME			VARCHAR(20) = ''
	 ,  @YYYYMM				VARCHAR(14) --쿼리내 YYYYMM추가 (FRONT 월 버튼사이 TEXT에서 넘어온정보)
)
AS
BEGIN

-- EXEC SP_WHR316_01_LIST '000001','','','',''
	SELECT  
		T1.EMPLOYEE_NO
		, T1.BASE_NAME 
		, T2.WG_CODE
		, T3.WG_NAME 
		, T1.DEPARTMENT_NAME 
		, T1.LEVEL_NAME 
		, T1.BASE_NAME
	FROM (
			SELECT	A.EMPLOYEE_NO
				,	A.BASE_NAME
				,	A.DEPARTMENT_CODE
				,	B.DEPARTMENT_NAME
				,	ISNULL(C.CODE_NAME_FULL,'') LEVEL_NAME
				,	B.VIEW_SEQ AS DEPARTMENT_SEQ 
				,	C.VIEW_SEQ 
			FROM TIN114 AS A
				LEFT JOIN TIN122 AS B
				ON A.DEPARTMENT_CODE = B.Department_Code 
				LEFT JOIN TCO101 C
				ON A.LEVEL_CODE = C.CODE_ID2 
				AND C.CODE_ID1 = '207'
			WHERE A.FACTORY_CODE = @FACTORY_CODE
			AND A.EMPLOYEE_NO <> '0000'
			AND ISNULL(A.REAL_USER_GBN, '0') = '1'
			AND (A.RETIRE_DATE='' OR A.RETIRE_DATE  IS NULL)
			AND	(@PAY_GROUP_CODE = '' OR (@PAY_GROUP_CODE <> '' AND A.PAY_GROUP_CODE = @PAY_GROUP_CODE))
			AND	(@PAY_TYPE_CODE = '' OR (@PAY_TYPE_CODE <> '' AND A.PAY_TYPE_CODE = @PAY_TYPE_CODE))
		) T1
		LEFT OUTER JOIN (
			SELECT FACTORY_CODE, EMPLOYEE_NO, WG_CODE, WG_FROM_DATE  
			FROM  TIN312W
			WHERE LEFT(WG_FROM_DATE, 6) = @YYYYMM  -- 조인 조건 추가/등록된 날짜를 년 월까지만 받아
		) T2 ON (T1.EMPLOYEE_NO = T2.EMPLOYEE_NO) -- 받아온 년 월과 같은것들만 SELECT하기
		LEFT OUTER JOIN (
			SELECT FACTORY_CODE , WG_CODE, WG_NAME FROM TIN310
		) AS T3 ON (T2.FACTORY_CODE = T3.FACTORY_CODE   AND T2.WG_CODE = T3.WG_CODE  )
	WHERE 1=1
	AND	(@WG_CODE = '' OR (@WG_CODE <> '' AND T2.WG_CODE = @WG_CODE))
	AND	(@DEPT_NAME = '' OR (@DEPT_NAME <> '' AND T1.Department_Name  = @DEPT_NAME))
	AND	(@BASE_NAME = '' OR (@BASE_NAME <> '' AND T1.BASE_NAME LIKE @BASE_NAME))
	ORDER BY T1.DEPARTMENT_SEQ, T1.VIEW_SEQ, T1.BASE_NAME 

END

```

## 문제점

조회 필터링 기능 자체는 정상적으로 동작하지만 IUD 등록과정에 문제가 있습니다.

```sql
USE [iPlusERP_IPACK_DEV]
GO
/****** Object:  StoredProcedure [dbo].[SP_WHR316_01_IUD]    Script Date: 2025-09-10 오전 9:30:14 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO


ALTER PROC	[dbo].[SP_WHR316_01_IUD]
(
		@FACTORY_CODE	VARCHAR(6)
	,	@EMPLOYEE_NO	VARCHAR(30)
	,	@WG_CODE		VARCHAR(6)
	,	@OPMAN_CODE		VARCHAR(10)


)

AS
BEGIN

	UPDATE TIN312W 
	SET 
			WG_CODE			=	@WG_CODE
		,	OPMAN_CODE		=	@OPMAN_CODE
		,	OPTIME			=	CONVERT(VARCHAR(8), GETDATE(), 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),4)
		,	WG_FROM_DATE	=	CONVERT(VARCHAR(8), GETDATE(), 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),0)
		,	WG_TO_DATE		=	CONVERT(VARCHAR(8), GETDATE() + 7, 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),0)
	WHERE	FACTORY_CODE	=	@FACTORY_CODE
	AND		EMPLOYEE_NO		=	@EMPLOYEE_NO
	
	IF	@@ROWCOUNT = 0 
		BEGIN

			INSERT INTO TIN312W
			( 
				FACTORY_CODE 
				, EMPLOYEE_NO 
				, WG_CODE
				, WG_BIGO
				, OPMAN_CODE
				, OPTIME
				, WG_FROM_DATE
				, WG_TO_DATE
			) 
			VALUES 
			( 
				@FACTORY_CODE
				,	@EMPLOYEE_NO
				,	@WG_CODE
				,	''
				,	@OPMAN_CODE
				,	CONVERT(VARCHAR(8), GETDATE(), 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),4)
				,	CONVERT(VARCHAR(8), GETDATE(), 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),0)
				,	CONVERT(VARCHAR(8), GETDATE() + 7, 112) + left(REPLACE(CONVERT(VARCHAR(8), GETDATE(), 108), ':', ''),0)
			) 
		END


END


```

해당 방법처럼 주차 정보에 대한 COMBO 구성과 주차 정보 등록 스토어나 콤보도 아직 만들지 않아

해당 절처럼 임의 추가 하는 방식 때문에 테이블 내부에 9월에 해당되는 데이터밖에 없습니다.

위 사항에 대한 수정안은 주차별 근무조 기능을 구현하면서 병행하겠습니다.
