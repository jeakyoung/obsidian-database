USE [iPlusERP_IPACK_DEV]

ALTER PROC	[dbo].[SP_WHR316_01_LIST]
(
		@FACTORY_CODE		VARCHAR(6)
	 ,	@PAY_GROUP_CODE		VARCHAR(10)
	 ,	@PAY_TYPE_CODE		VARCHAR(10)
	 ,	@WG_CODE			VARCHAR(6)
	 ,  @DEPT_NAME			VARCHAR(10)
	 ,	@BASE_NAME			VARCHAR(20) = ''
	 ,  @WG_FROM_DATE		VARCHAR(8)
)
AS
BEGIN
	;WITH
	DATE_CTE AS (
        SELECT 
            A.FACTORY_CODE,
			A.EMPLOYEE_NO,
			A.WG_CODE,
			CONVERT(VARCHAR(8), DATEADD(DAY, 7, CONVERT(DATE, A.WG_FROM_DATE, 112)), 112) AS WG_FROM_DATE
        FROM TIN312W A
        WHERE A.FACTORY_CODE = @FACTORY_CODE
		AND A.WG_FROM_DATE = CONVERT(VARCHAR(8), DATEADD(DAY, -7, CONVERT(DATE, @WG_FROM_DATE, 112)), 112)
    )
	
-- EXEC SP_WHR316_01_LIST '000001','','','',''
	SELECT  
		  T1.EMPLOYEE_NO
		, T1.BASE_NAME 
		, T2.WG_CODE
		, T2.WG_CODE2
		, T3.WG_NAME 
		, T1.DEPARTMENT_NAME 
		, T1.LEVEL_NAME 
		, T2.WG_FROM_DATE
		, T2.WG_TO_DATE
	FROM (
			SELECT	
					A.EMPLOYEE_NO
				,	A.BASE_NAME
				,	A.DEPARTMENT_CODE
				,	B.DEPARTMENT_NAME
				,	ISNULL(C.CODE_NAME_FULL,'') LEVEL_NAME
				,	B.VIEW_SEQ AS DEPARTMENT_SEQ 
				,	C.VIEW_SEQ 
			    FROM TIN114 AS A
				LEFT JOIN TIN122 AS B
				ON  A.DEPARTMENT_CODE = B.Department_Code 
				LEFT JOIN TCO101 C
				ON  A.LEVEL_CODE = C.CODE_ID2 
				AND C.CODE_ID1 = '207'
			WHERE A.FACTORY_CODE = @FACTORY_CODE
			--AND A.EMPLOYEE_NO <> '0000'
			AND ISNULL(A.REAL_USER_GBN, '0') = '1'
			AND (A.RETIRE_DATE   = '' OR A.RETIRE_DATE  IS NULL)
			AND	(@PAY_GROUP_CODE = '' OR (@PAY_GROUP_CODE <> '' AND A.PAY_GROUP_CODE = @PAY_GROUP_CODE))
			AND	(@PAY_TYPE_CODE  = '' OR (@PAY_TYPE_CODE <> '' AND A.PAY_TYPE_CODE = @PAY_TYPE_CODE))
		) T1
		LEFT OUTER JOIN (
				SELECT E.FACTORY_CODE
					 , E.EMPLOYEE_NO
				     , E.WG_CODE AS WG_CODE
					 , D.WG_CODE AS WG_CODE2
					 , E.WG_FROM_DATE
					 , E.WG_TO_DATE
			FROM  TIN312W E
			LEFT JOIN DATE_CTE D
			ON D.FACTORY_CODE = E.FACTORY_CODE
			AND D.EMPLOYEE_NO = E.EMPLOYEE_NO
			WHERE E.WG_FROM_DATE = @WG_FROM_DATE
		) T2 ON (T1.EMPLOYEE_NO = T2.EMPLOYEE_NO)
		LEFT OUTER JOIN (
			SELECT FACTORY_CODE
				 , WG_CODE
				 , WG_NAME
			FROM  TIN310
		) AS T3 ON (T2.FACTORY_CODE = T3.FACTORY_CODE AND T2.WG_CODE = T3.WG_CODE  )
	WHERE 1=1
	AND	(@WG_CODE   = '' OR (@WG_CODE <> '' AND T2.WG_CODE = @WG_CODE))
	AND	(@DEPT_NAME = '' OR (@DEPT_NAME <> '' AND T1.Department_Name  = @DEPT_NAME))
	AND	(@BASE_NAME = '' OR (@BASE_NAME <> '' AND T1.BASE_NAME LIKE '%'+@BASE_NAME+'%'))
	ORDER BY T1.DEPARTMENT_SEQ, T1.VIEW_SEQ, T1.BASE_NAME 

END
