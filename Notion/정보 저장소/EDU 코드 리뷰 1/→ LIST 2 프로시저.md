```sql
ALTER PROCEDURE [dbo].[SP_HHR102_LIST]
@FACTORY_CODE   VARCHAR(6) = NULL,
@WORK_CODE      VARCHAR(6) = NULL,
@LINE_CODE      VARCHAR(3) = NULL,
@LINE_NAME      VARCHAR(8) = NULL,
@VIEW_SEQ       INT        = NULL,
@DELETE_FLAG    VARCHAR(1) = NULL
AS
BEGIN
SET NOCOUNT ON;
SELECT
      B.FACTORY_CODE
    , B.WORK_CODE
    , A.LINE_CODE
    , A.LINE_NAME
    , A.VIEW_SEQ
    , B.DELETE_FLAG
FROM  HHR102 A
LEFT JOIN HHR101 B ON A.WORK_CODE = B.WORK_CODE
WHERE (B.FACTORY_CODE = @FACTORY_CODE OR @FACTORY_CODE IS NULL)
  AND (B.WORK_CODE    = @WORK_CODE    OR @WORK_CODE IS NULL)
  AND (A.LINE_CODE    = @LINE_CODE    OR @LINE_CODE IS NULL)
  AND (A.LINE_NAME    = @LINE_NAME    OR @LINE_NAME IS NULL)
  AND (A.VIEW_SEQ     = @VIEW_SEQ     OR @VIEW_SEQ IS NULL)
  AND (B.DELETE_FLAG  = @DELETE_FLAG  OR @DELETE_FLAG IS NULL)
END
```
