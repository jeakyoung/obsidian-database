```sql
ALTER PROCEDURE [dbo].[SP_HHR102_IUD]
    @FACTORY_CODE           VARCHAR(6),
    @WORK_CODE              VARCHAR(6),
    @LINE_CODE              VARCHAR(3),
    @LINE_NAME              VARCHAR(50),
    @VIEW_SEQ               INT,             --Int 형식선언
    @PROD_LIMIT_QTY         FLOAT,           --Float 형식선언
    @OPMAN_CODE             VARCHAR(4),
    @LINE_INSP_DATE         VARCHAR(24),
    @IMPORTANT_LINE_FLAG    VARCHAR(1),
    @IUD                    VARCHAR(2)
AS
BEGIN
																		 
																		  .
																			.
																	    .
																	    
         BEGIN
            IF ( @LINE_CODE = '' AND @VIEW_SEQ = '' )
                BEGIN
              SELECT @LINE_CODE = RIGHT('000' + CAST(CAST(ISNULL(MAX(LINE_CODE),'0') AS INT) + 1 AS VARCHAR(3)),3),
                     @VIEW_SEQ  = ISNULL(MAX(VIEW_SEQ),　0) + 1 
                     -- 불필요한 CAST문 삭제
                    FROM   HHR102
                    WHERE  WORK_CODE  = @WORK_CODE
                END
																		  
																		  .
																			.
																	    .      
																	                
    DECLARE     @sum   FLOAT                          --처음부터 Float로 시작
               ,@count INT;                           --처음부터 Int로 시작
        SELECT  @sum =  ISNULL(SUM( PROD_LIMIT_QTY ),0),   -- Cast문 전체삭제
                @count = COUNT(*)                          -- Cast문 전체삭제
        FROM HHR102
        WHERE FACTORY_CODE      = @FACTORY_CODE
            AND WORK_CODE       = @WORK_CODE
            AND ISNULL(DELETE_FLAG, '0') <> '1'

        UPDATE HHR101
            SET   PROD_LIMIT_QTY    = @sum
                 ,LINE_COUNT        = @count
            WHERE FACTORY_CODE      = @FACTORY_CODE
                  AND WORK_CODE     = @WORK_CODE
END
```
