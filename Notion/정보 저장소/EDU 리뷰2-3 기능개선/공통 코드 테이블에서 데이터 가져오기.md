Store 파일 내 공통적으로 사용하는 형식 ( GetTCO101 )을 호출했습니다.

```javascript
onSetCombo01: function (ITEM_KIND1, ITEM_KIND2) {
    var me = this;
    var vm = me.getView().getViewModel();
    var store = vm && vm.getStore('store_combo');
    if (!store) return;
    var cfg = {};
    if (ITEM_KIND1 || ITEM_KIND2) {
      cfg.params = { ITEM_KIND1: ITEM_KIND1, ITEM_KIND2: ITEM_KIND2 };
    }
    store.load(cfg);
  },
```

onSetCombo 동작으로 combo내 정보를 cfg 파라미터에 담아 store에 Load 합니다.

해당 부는 Init 동작마다 호출합니다.

```javascript
onRenderer_02: function (value, meta, record) {
		var me = this;

		var store = this.getViewModel().getStore('store_combo');
		if (value) {
			var record = store.findRecord('CODE_ID2', value);
			if (record) {
				return record.get('CODE_NAME_FULL');
			}
		}
	},
```

콤보버튼 랜더러로 record에 CODE_ID2로 되어있는 record를 찾아와서 해당하는 ID에 맞는 NAME_FULL
